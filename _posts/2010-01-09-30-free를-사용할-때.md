void deleteNode(NODE* hd, int delCnt){  
  
int i;  
NODE* delPt;

for(i=1;i<delCnt;i++){ // delCnt만큼 노드순회후 삭제  
hd = hd->next;  
}  
delPt = hd->next;  
hd->next = hd->next->next;  
  
free(delPt);  
}
