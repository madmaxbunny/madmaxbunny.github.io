public class Str01 {

 public static void main(String[] args) {

  if (i == j) {

  } else {
   System.out.println("다름");
  }

  System.out.println("이름을 입력하세요");
  Scanner sc = new Scanner(System.in);
  String s1 = sc.next();
  System.out.println(s1 + "님 반가워요");
  
  sc.close(); // 닫아줘야 버퍼가 차지 않는다.
 }

}
