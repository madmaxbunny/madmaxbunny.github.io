## 2. 핵심 설명 및 설정 가이드

#include <stdio.h>  
#include <stdlib.h>

// 이중 동적배열 예제  
void main(){

int i;  
/* p[5][3] 만들기 */

int **p; // index 변수 선언

p = ( int** )malloc( sizeof( int* ) * 5 ); // 만들 이중 배열의 행을 5만큼 넉넉히 잡는다.  
/*  
p[0][]  
p[1][]  
p[2][]  
p[3][]  
p[4][]  
*/

for(i=0;i<5;i++){  
p[i] = (int*)malloc(sizeof(int)*3);  
  
/* p[i][0], p[i][1], p[i][2] 과정을 반복 */  
  
}

p[4][2] = 3;

}
